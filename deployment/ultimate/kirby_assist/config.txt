# This file changes startup settings without recompiling Rust. Restart Smash.
# Accepted keys and ranges are defined in LiveConfig::parse in live_input.rs.
# To add a new setting, extend the Rust field/default/parser and its tests first;
# adding an arbitrary key here makes config parsing fail rather than adding a tool.
# Kirby Assist: Pro Controller direct raw controller-memory button input build.
# Read once when Smash starts. Restart the game after editing.
# Editing workflow: change one value, save, copy this file to the Switch at
# SD:/ultimate/kirby_assist/config.txt, fully close Smash, then reopen to reload.
# Rust edits require a rebuild and copying the new NRO; this file needs no build.
# Values below override Rust defaults. Deleting a key restores its Rust default,
# which may differ from this supplied file (especially turbo/ledge/takeover).
enabled = true
auto_mash = true
auto_tech = true
# Diagnostic build only: record gate reasons and actual ZR press edges.
# Silent compiles the logger out regardless of this setting.
tech_debug = true

# First Switch controller is 0; second is 1, etc. Pro Controller only.
npad_id = 0

# Auto mash uses the left stick: Right -> Up -> Left -> Down, one full-strength
# direction per distinct mashable mapper update. It generates no mash buttons.
# takeover suppresses competing physical action buttons and C-stick while mashing;
# during tech it still owns only ZR. Physical control returns when the tool ends.
takeover = true
# Old alternate_mash / mash_buttons_per_frame / mash_interval_frames entries are
# ignored for compatibility. Stick mashing has no added interval/release gap.

# Predict stage contact this many motion frames ahead (1 to 5).
tech_lookahead_frames = 5

# Minimum observed input updates between automated ZR presses.
# This does not change the game's actual tech lockout.
tech_cooldown_frames = 40

# Hold D-pad Left/Right: crouch/release with a slight movement that direction.
# Reserves side taunts during valid gameplay; both directions held cancel movement.
# Diagonals involved in a macro press are consumed through release.
dpad_turbo = true
# Observed mapper updates per crouch/release phase (1 to 10).
turbo_phase_frames = 2
# Normalized horizontal stick strength during the move phase (0.05 to 0.6).
turbo_walk_strength = 0.6

# Click L3 to toggle ledge assist: prepare at the nearest detected stage ledge.
# Requires A=Attack, X=Jump, R=Grab, ZR=Shield, Tap Jump OFF.
# Ground-only: waits in air, selects the current floor edge on landing, then positions.
# Click L3 again to return physical control. Timing/geometry still limit coverage.
ledge_trap = true
# Witnessed ledge drop -> jump/aerial: stop moving and shield; no blind grab.
# Active while L3 mode is toggled on. Click L3 again for a manual punish.
ledge_drop_defense = true
# Tap D-pad Down: find the nearest floor edge, run off, Down, then inward to catch.
# No opponent is required. Enabled by default even when this key is omitted.
# The sequence ends at ledge catch; choose your own follow-up. L3 is separate.
auto_ledge_trump = true
# Normal get-up only: down_tilt (low poke) or forward_tilt (angled down).
# Jump uses short-hop neutral air; roll/get-up-attack follow-ups still use Grab.
ledge_getup_attack = forward_tilt

# Jump response: neutral X+A timed for the predicted neutral-air active window.
# Root-position gates, not exact hitboxes: range 1..20, height 1..40.
ledge_anti_air_range = 8.0
ledge_anti_air_height = 18.0

# Initial opponent acquisition limits (world units, finite 1..20).
ledge_trap_range = 20.0
ledge_trap_vertical_range = 10.0
# Separate proximity gates for normal-getup down tilt and roll/shield grabs.
ledge_tilt_range = 15.0
ledge_grab_range = 9.0

# Continue following acquired opponents outside the initial box (1..100).
ledge_follow_range = 40.0
ledge_follow_vertical_range = 48.0
# Ground-probed stance and pursuit; default 8 units inward leaves more room
# to react to rolls. Momentum prediction reduces stance overshoot. This offset
# is a tunable starting point, not a universal coverage position.
ledge_auto_position = true
ledge_stand_distance = 10.0
# Roll endpoint distance inward from the ledge: auto = estimate ONCE from motion.
# A measured numeric value (2..100) commits immediately when the roll is read.
# This is travel distance, not grab reach; character-specific tuning is required.
ledge_roll_distance = auto
ledge_move_strength = 0.85

# One early Grab pulse near shieldstun end when close (0..6).
ledge_grab_buffer_frames = 3
# Button actually mapped to Grab: R/L/ZL. ZR remains Shield.
# Retries require a release; native startup pauses them and capture ends them.
# Shield stays held through block hitstop. Pushback pursuit drops it once and
# waits through guard-off before moving/grabbing, without raising Shield again.
ledge_grab_button = R
# After a confirmed grab, choose a horizontal throw toward the remembered ledge.
# Waits for native catch-wait; never presses A for pummels. Requires acquiring
# the cliff anchor while hanging. Set false to keep manual throw selection.
ledge_throw_outward = true
