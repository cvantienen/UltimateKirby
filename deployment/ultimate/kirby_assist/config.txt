# This file changes startup settings without recompiling Rust. Restart Smash.
# Accepted keys and ranges are defined in LiveConfig::parse in live_input.rs.
# To add a new setting, extend the Rust field/default/parser and its tests first;
# adding an arbitrary key here makes config parsing fail rather than adding a tool.
# Kirby Assist: direct raw controller-memory button input build.
# Read once when Smash starts. Restart the game after editing.
# Editing workflow: change one value, save, copy this file to the Switch at
# SD:/ultimate/kirby_assist/config.txt, fully close Smash, then reopen to reload.
# Rust edits require a rebuild and copying the new NRO; this file needs no build.
# Values below override Rust defaults. Deleting a key restores its Rust default,
# which may differ from this supplied file (especially turbo/ledge/takeover).
enabled = true
auto_mash = true
auto_tech = true
# Diagnostic: extra tech details plus ledge wait snapshots at most twice/second.
# Action changes and attack edges remain logged when false. Background I/O is buffered.
# Silent compiles the logger out regardless of this setting.
tech_debug = true

# Local physical controller: 0 = Player 1, 1 = Player 2, etc. No style filter.
# Follows that controller's actual mapper slot; this is not your Arena battle position.
# All human-controlled fighters use the same inputs in every battle mode.
# Move timing/range defaults are still tuned for Kirby.
player_index = 0

# Auto mash uses the left stick: Right -> Up -> Left -> Down, one full-strength
# direction per distinct mashable mapper update. It generates no mash buttons.
# takeover suppresses competing physical action buttons and C-stick while mashing;
# during tech it releases shoulder buttons as needed, then requests ZR.
# Physical control returns when the tool ends.
takeover = true
# Old alternate_mash / mash_buttons_per_frame / mash_interval_frames entries are
# ignored for compatibility. Stick mashing has no added interval/release gap.

# Base stage-contact lookahead (1 to 5 game frames). Learned input delay extends it.
tech_lookahead_frames = 5

# Cooldown starts only after a fresh mapped shield press, excluding hitstop.
# An observed successful tech clears our cooldown; native lockout is unchanged.
# Grounded Ganon Side B uses its linked capture clock; the aerial slam is untechable.
tech_cooldown_frames = 40

# Side D-pad: tap for one short-hop/downward-angled-air-dodge attempt; hold to repeat.
# The first airborne native frame must permit the dodge; otherwise it aborts.

# Hold L3: creep in current facing while alternating crouch/release.
# L3 is turbo only. Hold Y separately for ledge assist; punish inputs take priority.

# Observed native animation updates per crouch/release phase (1 to 10).

# Normalized creep strength (L3 caps this to 0.45).

# Directional attack recipes temporarily override the cap only when eligible.
# Only its own jump punish retains aerial input ownership.

ledge_drop_defense = true

# Normal get-up: neutral_air starts its hitbox about 3 frames before protection ends.

# Jump response: prepare the hop during launch, then timed neutral-stick+A in air.
